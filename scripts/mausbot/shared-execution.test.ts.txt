import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import type { RuntimeEvent, SendTurnInput } from "../contracts.ts";
import { mountChatTools } from "./chat-mcp-tools.ts";
const tools = vi.hoisted(() => ({
  execute: vi.fn(async () => ({ ok: true, text: "Original bot tool result" })),
  close: vi.fn(async () => {}),
  validate: vi.fn(),
}));
vi.mock("./chat-mcp-tools.ts", () => ({
  mountChatTools: vi.fn(async () => ({
    definitions: [{ type: "function", function: { name: "agents_team_post", description: "Post to the engineering room", parameters: { type: "object" } } }],
    ...tools,
    view: () => ({ title: "Team post", input: { text: "Ready" }, ask: true }),
  })),
}));
import { J1SharedDriver, decodeSharedModel, encodeSharedModel, nativeResumeCursor, savedHistoryForTurn, savedIncidentReports } from "./j1-shared.ts";

const request = { instanceId: "j1-shared", displayName: "J1", enabled: true, environment: {}, config: {} };
const turn: SendTurnInput = { botId: "bot", threadId: "thread", j1BotName: "Ada", j1ProjectId: "project",
  text: "Build the change", model: encodeSharedModel("codex-work", "gpt-model"), system: "Bot's saved instructions",
  approvalMode: "ask", cwd: "C:/workspace", images: [], transcript: [{ role: "user", text: "Prior instruction" }],
};
let calls: Array<Record<string, unknown>>;
let instances: Array<Awaited<ReturnType<typeof J1SharedDriver.create>>>;
let toolResult: (value: Record<string, unknown>) => void;
let oldSend: typeof process.send;
let oldConnected: boolean | undefined;
let catalogUnavailable = false;
let wrongStartAcknowledgment = false;
const event = (value: unknown) => process.emit("message", { type: "j1:execution-event", event: value }, undefined);
const currentTurnId = () => calls.findLast(call => call.action === "start")?.turnId;
const runtime = (type: string, payload: object, extra = {}, turnId = currentTurnId()) =>
  event({ kind: "runtime", botId: "bot", threadId: "thread", turnId,
  event: { eventId: "event", provider: "codex", threadId: "canonical", createdAt: "2026-11-18T00:00:00Z", type, payload, ...extra } });

beforeEach(() => {
  catalogUnavailable = false;
  wrongStartAcknowledgment = false;
  calls = []; instances = []; toolResult = () => {};
  vi.mocked(mountChatTools).mockClear();
  tools.execute.mockClear(); tools.close.mockClear(); tools.validate.mockReset();
  oldSend = process.send; oldConnected = process.connected;
  Object.defineProperty(process, "connected", { value: true, configurable: true });
  process.send = ((message: { requestId: string; input: Record<string, unknown> }, callback?: (error: Error | null) => void) => {
    calls.push(message.input);
    const value = message.input.action === "catalog"
      ? [{ instanceId: "codex-work", driver: "codex", displayName: "Work", enabled: true, installed: true, auth: { status: "authenticated" }, models: [{ slug: "gpt-model", name: "GPT", isDefault: true }] }]
      : message.input.action === "start"
        ? { turnId: wrongStartAcknowledgment ? "different-turn" : message.input.turnId }
        : {};
    if (message.input.action === "tool-result") toolResult(message.input);
    queueMicrotask(() => process.emit("message", { type: "j1:execution-result", requestId: message.requestId,
      ...(catalogUnavailable && message.input.action === "catalog" ? { error: "Shared execution is unavailable." } : { value }) }, undefined));
    callback?.(null);
    return true;
  }) as typeof process.send;
});
afterEach(async () => {
  for (const instance of instances) await instance.dispose();
  process.send = oldSend;
  Object.defineProperty(process, "connected", { value: oldConnected, configurable: true });
});
async function fixture() {
  const instance = await J1SharedDriver.create(request);
  instances.push(instance);
  await instance.startupModelRefresh;
  return instance;
}

describe("J1 shared bot adapter", () => {
  it("admits a tool-free consultation and rejects action calls without mounting tools", async () => {
    const instance = await fixture();
    const events: RuntimeEvent[] = [];
    instance.adapter.onEvent(event => events.push(event));
    const result = await instance.adapter.sendTurn({ ...turn, j1Consultation: true });
    expect(calls.find(call => call.action === "start")?.tools).toEqual([]);
    expect(mountChatTools).not.toHaveBeenCalled();
    const denied = Promise.withResolvers<Record<string, unknown>>();
    toolResult = denied.resolve;
    event({ kind: "tool-call", botId: "bot", threadId: "thread", turnId: currentTurnId(), requestId: "consultation-tool",
      name: "agents_team_post", arguments: { text: "Blocked" } });
    expect(await denied.promise).toMatchObject({ value: { ok: false } });
    expect(tools.execute).not.toHaveBeenCalled();
    runtime("turn.completed", { state: "completed" }, {}, "stale-turn");
    expect(instance.adapter.hasSession("thread")).toBe(true);
    expect(events).toEqual([]);
    runtime("turn.completed", { state: "completed" }, {}, result.turnId);
    expect(events.at(-1)).toMatchObject({ type: "turn.completed", turnId: result.turnId, ok: true });
    expect(instance.adapter.hasSession("thread")).toBe(false);
    expect(tools.close).not.toHaveBeenCalled();
    await instance.adapter.sendTurn(turn);
    expect(mountChatTools).toHaveBeenCalledOnce();
    expect(calls.findLast(call => call.action === "start")?.tools).toEqual([
      expect.objectContaining({ name: "agents_team_post" }),
    ]);
  });
  it("rejects an acknowledgment for a different turn and releases the original reservation", async () => {

    expect(instance.adapter.hasSession("thread")).toBe(false);
    expect(tools.close).toHaveBeenCalledOnce();
    wrongStartAcknowledgment = false;
    expect(await instance.adapter.sendTurn(turn)).toMatchObject({
      turnId: calls.findLast(call => call.action === "start")?.turnId,
    });
  });

  it("continues a saved computer choice without sending the old user line again", async () => {
    const instance = await fixture();
    const choice = { id: "accepted", sourceGeneration: "root", dispatchGeneration: "selected-turn",
      selected: "browser", previousSurface: "local", sourceSurface: "local" } as const;
    const source = { id: "original-user", role: "user" as const, kind: "text", at: 10, text: "My request",
      j1CoordinatorRootId: choice.sourceGeneration, j1ComputerSelections: [choice],
      attachments: [{ path: "/api/attachments/original.txt", kind: "file", mime: "text/plain" }] };
    const reply = { id: "settled-reply", kind: "text", role: "bot" as const, at: 11, text: "Choose a computer",
      j1ProviderMessageId: "saved-reply" };
    const anchor = { messageId: source.id, promptText: "Continue using the selected browser", currentImages: [],
      origin: { root: true as const, nodeId: "root", requestId: "root", execution: 2, restarts: 0,
        dispatch: "computer" as const, computerSelection: choice, sourceMessageId: source.id,
        approvalMode: "ask" as const, restrictedTools: false, current: () => true },
    };
    const saved = [source, reply];
    expect(savedHistoryForTurn(saved, anchor, { managed: "." })).toMatchObject({
      promptText: anchor.promptText, promptAttachments: [],
      messages: [{ nativeMessageId: source.id, role: "user", text: source.text,
        attachments: [{ root: "managed", filename: "original.txt" }] },
      { nativeMessageId: reply.id, role: "assistant", mirrorMessageId: "saved-reply" }],
      origin: { computerSelection: choice },
    });
    expect(() => savedHistoryForTurn([{ ...source, j1ComputerSelections: undefined }, reply], anchor, { managed: "." })).toThrow();
    expect(() => savedHistoryForTurn([{ ...source, j1ComputerSelections: [{ ...choice, dispatchGeneration: "stale" }] }, reply],
      anchor, { managed: "." })).toThrow();
    expect(() => savedHistoryForTurn([...saved, { ...source, id: "newer" }], anchor, { managed: "." })).toThrow();
    const next = { id: "next", previousChoiceId: choice.id, sourceGeneration: choice.dispatchGeneration,
      dispatchGeneration: "next-turn", selected: "vm", previousSurface: "browser", sourceSurface: "browser" } as const;
    const switched = [{ ...source, j1ComputerSelections: [choice, next] }, reply];
    const nextAnchor = { ...anchor, promptText: "Continue in the VM",
      origin: { ...anchor.origin, execution: 3, computerSelection: next } };
    expect(savedHistoryForTurn(switched, nextAnchor, { managed: "." })).toMatchObject({
      promptText: nextAnchor.promptText, promptAttachments: [], origin: { computerSelection: next },
      messages: [{ nativeMessageId: source.id, role: "user" }, { nativeMessageId: reply.id, role: "assistant" }],
    });
    expect(savedHistoryForTurn(switched, anchor, { managed: "." }).origin?.computerSelection).toEqual(choice);
    expect(() => savedHistoryForTurn([{ ...source, j1ComputerSelections: [next] }, reply],
      nextAnchor, { managed: "." })).toThrow();
    await instance.adapter.sendTurn({ ...turn, text: anchor.promptText, transcript: undefined, images: [], j1SavedHistoryAnchor: anchor });
    expect(calls.find(call => call.action === "start")).toMatchObject({
      nativeOriginStart: true, text: anchor.promptText, sharedContinuation: false, images: [],
    });
    await instance.adapter.interruptTurn("thread");
    expect(calls.find(call => call.action === "stop")).toMatchObject({ botId: "bot", threadId: "thread" });
  });

  it("retains the coordinator request as assistant history and excludes history from current input", async () => {
    const instance = await fixture();

    const saved = [
      { id: "older", role: "user" as const, kind: "text", text: "Prior line", at: 10 },

      ["older", "user", "Prior line"], ["request", "assistant", "Assigned work"],

    const instance = await fixture();

    runtime("turn.completed", { state: "completed" }, {}, previous.turnId);
    const current = await instance.adapter.sendTurn({ ...turn, approvalMode: "full" });
    const owner = { botId: "bot", threadId: "thread" };
    const request = { ...owner, kind: "tool-call", requestId: "reused-request-id",
      name: "agents_team_post", arguments: {} };
    event({ ...request, turnId: previous.turnId });
    event(request);
    expect(tools.validate).not.toHaveBeenCalled();
    expect(tools.execute).not.toHaveBeenCalled();
    event({ ...owner, kind: "failed", turnId: previous.turnId, message: "Stale setup failure" });
    event({ ...owner, kind: "failed", message: "Unscoped setup failure" });
    expect(instance.adapter.hasSession("thread")).toBe(true);

    const started = Promise.withResolvers<AbortSignal>();
    const reply = Promise.withResolvers<Record<string, unknown>>();
    toolResult = reply.resolve;
    tools.execute.mockImplementationOnce(async (...args: unknown[]) => {
      const signal = args[2] as AbortSignal;
      started.resolve(signal);
      await new Promise<void>(resolve => signal.addEventListener("abort", () => resolve(), { once: true }));
      return { ok: false, text: "Cancelled" };
    });
    event({ ...request, turnId: current.turnId });
    const signal = await started.promise;
    const cancel = { ...owner, kind: "tool-cancel", requestId: request.requestId };
    event({ ...cancel, turnId: previous.turnId });
    event(cancel);
    expect(signal.aborted).toBe(false);
    event({ ...cancel, turnId: current.turnId });
    expect(signal.aborted).toBe(true);
    expect(await reply.promise).toMatchObject({ requestId: request.requestId, value: { ok: false } });
    expect(tools.execute).toHaveBeenCalledOnce();

    const events: RuntimeEvent[] = [];
    instance.adapter.onEvent(value => events.push(value));
    event({ ...owner, kind: "failed", turnId: current.turnId, message: "Current setup failure" });
    expect(events.at(-1)).toMatchObject({ type: "turn.completed", turnId: current.turnId, ok: false });
    expect(instance.adapter.hasSession("thread")).toBe(false);
  });

    runtime("session.exited", { reason: "CLI crashed" });
    expect(instance.adapter.hasSession("thread")).toBe(false);
    expect(events.at(-1)).toMatchObject({ type: "turn.completed", ok: false });
  });
});

it("reads the saved branch before the exact current message and keeps current files separate", () => {
  const root = { managed: "/owned-attachments" };
  const prior = { id: "one", at: 2000, role: "bot", kind: "text",
    text: "Older reply", turnCompletedAt: 2400,
    attachments: [{ kind: "file", path: "/api/attachments/old.txt", name: "old.txt", mime: "text/plain" }],
  } as const;
  const current = { id: "two", at: 3000, role: "user", kind: "text", text: "Current",
    attachments: [{ kind: "image", path: "/api/attachments/new.png", mime: "image/png" }],
  } as const;
  const anchor = { messageId: current.id, promptText: "Clean prompt",
    currentImages: [{ path: "/api/attachments/new.png", mime: "image/png" }],
  };
  const saved = savedHistoryForTurn([prior, current], anchor, root);
  expect(saved.messages.map(({ nativeMessageId, createdAt, completedAt, role }) =>
    [nativeMessageId, createdAt, completedAt, role])).toEqual([
    ["one", "1970-01-01T00:00:02.000Z", "1970-01-01T00:00:02.400Z", "assistant"],
  ]);
  expect(saved.messages[0]?.attachments[0]?.filename).toBe("old.txt");
  expect(saved.promptAttachments.map(item => item.filename)).toEqual(["new.png"]);
  expect(saved.promptText).toBe("Clean prompt");
  expect(() => savedHistoryForTurn([prior], anchor, root)).toThrow();
  expect(() => savedHistoryForTurn([prior, current, { ...current, id: "newer" }], anchor, root)).toThrow();
  expect(() => savedHistoryForTurn([prior, current], {
    ...anchor, currentImages: [{ path: "/api/attachments/foreign.png", mime: "image/png" }],
  }, root)).toThrow();
  expect(() => savedHistoryForTurn([prior, { ...current,
    attachments: [{ kind: "image", path: "../foreign.png", mime: "image/png" }],
  }], anchor, root)).toThrow();
});

describe("saved Chief incident reports", () => {
  const report = (id: string, sourceThreadId = "affected", recipientBotId = "chief") => ({
    id, at: 1, role: "user" as const, kind: "text", text: id,
    peerAsk: { botId: "reporter", name: "Reporter", unattended: true,
      incident: { threadId: sourceThreadId, j1Recipient: { botId: recipientBotId, threadId: "notices" } } },
  });
  const read = (rows: Parameters<typeof savedIncidentReports>[0], ids = ["one", "two"]) =>
    savedIncidentReports(rows, ids, "notices", "chief");

  it("returns the exact ordered batch with native provenance", () => {
    const rows = [report("one"), report("two", "other")];
    const batch = read(rows);
    expect(batch?.map(row => [row.nativeMessageId, row.sourceThreadId])).toEqual([["one", "affected"], ["two", "other"]]);
    expect(read(rows)).toEqual(batch);
    expect(read([report("one"), { ...rows[1]!, text: "changed" }])).not.toEqual(batch);
  });

  it("preserves native report rows and requires current generation ownership", () => {
    const rows = [report("one"), report("two")];
    let current = true;
    const origin = { nodeId: "incident-batch", requestId: "incident-batch", execution: 1,
      dispatch: "first" as const, restarts: 0, sourceMessageId: "two",
      approvalMode: "full" as const, restrictedTools: true, current: () => current,
      incidentBatch: { generation: "owned", reports: read(rows)! } };
    const history = () => savedHistoryForTurn(rows, {
      messageId: "two", promptText: "computed batch input", currentImages: [], origin,
    }, { managed: "unused" });
    expect(history().messages.map(row => [row.nativeMessageId, row.role])).toEqual([["one", "user"], ["two", "user"]]);
    expect(history().promptText).toBe("computed batch input");
    current = false;
    expect(history).toThrow("does not own this saved input");
  });

  it("rejects reordered, duplicate, mixed, foreign, and stopped reports", () => {
    const one = report("one");
    const two = report("two");
    expect(read([two, one])).toBeUndefined();
    expect(read([one, two], ["two", "two"])).toBeUndefined();
    expect(read([one, { ...two, peerAsk: undefined }])).toBeUndefined();
    expect(read([one, report("two", "notices")])).toBeUndefined();
    expect(read([one, report("two", "affected", "another-chief")])).toBeUndefined();
    expect(read([one, { ...two, requestCancelled: true }])).toBeUndefined();
    expect(read([one, two, { ...report("new"), peerAsk: undefined }])).toBeUndefined();
  });
});
