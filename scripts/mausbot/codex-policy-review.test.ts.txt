  it.each([false, true].flatMap(resumed => ["allow", "deny"].map(behavior => ({ resumed, behavior }))))(
    "keeps Full command review manual: %j", async ({ resumed, behavior }) => {
      process.env.FAKE_CODEX_APPROVAL_REQUEST = JSON.stringify({
        method: "item/commandExecution/requestApproval",
        params: { threadId: "codex-thread-1", turnId: "turn-1", command: "Write-Output fixture", cwd: scratch },
      });
      await create({ mode: "approval", fullAuto: true });
      await instance.adapter.sendTurn({ threadId: "manual-review", text: "Fixture only", approvalMode: "full",
        ...(resumed ? { resumeCursor: "codex-thread-1" } : {}) });
      const opened = await recorder.until(event => event.type === "request.opened");
      if (opened.type !== "request.opened") throw new Error("Missing manual command approval");
      expect(opened.requiresExplicitApproval).toBe(true);
      expect(recorder.events.some(event => event.type === "request.resolved")).toBe(false);
      await instance.adapter.respondToRequest("manual-review", opened.requestId, { behavior: behavior as "allow" | "deny" });
      await recorder.until(event => event.type === "turn.completed");
      expect(recorder.events.find(event => event.type === "request.resolved")).toMatchObject({ behavior });
    },
  );
