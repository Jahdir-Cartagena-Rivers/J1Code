import { describe, expect, it, vi } from "vitest";
import { availableTools, catalogProfileFromEnv } from "../drivers/agents-catalog.ts";
import { callTool, type ToolCallContext } from "../drivers/agents-call.ts";
import { routingToolAllowed } from "./agents-routing.ts";
describe("native shared routing SDK", () => {
  it("advertises native project discovery and routing through the original provider catalog", () => {
    const names = availableTools(catalogProfileFromEnv({ OMB_BOT_ID: "crown" })).map(tool => tool.name);
    expect(names).toEqual(expect.arrayContaining(["list_chat_projects", "routing_mode", "claim_task", "release_task", "route_task", "routing_status", "list_routes", "acknowledge_route"]));
  });
  it("uses native project discovery from delegated turns and rejects injected identity", async () => {
    const apiResponse = vi.fn(async () => ({ ok: true, status: 200, body: [{ id: "j1", title: "J1 Code", workspaceRoot: "C:/Projects/J1_Code" }] }));
    const context = { botId: "builder", threadId: "work", depth: 1, externalRuntime: false, sharedComputers: false, client: { api: async () => ({}), apiResponse } } as ToolCallContext;
    expect(JSON.parse((await callTool("list_chat_projects", {}, context)).text)).toEqual([{ id: "j1", title: "J1 Code", workspaceRoot: "C:/Projects/J1_Code" }]);
    expect((await callTool("list_chat_projects", { botId: "forged" }, context)).isError).toBe(true);
    expect(apiResponse).toHaveBeenCalledTimes(1);
  });
  it("uses the actual capability-bound source and rejects actor/approval overrides", async () => {
    const apiResponse = vi.fn(async () => ({ ok: true, status: 200, body: { status: "suggested" } }));
    const context: ToolCallContext = { botId: "crown", threadId: "main", depth: 0, externalRuntime: false, sharedComputers: false, client: { api: async () => ({}), apiResponse } };
    const args = { taskId: "build", requestId: "stable", request: "Updated request" };
    const result = await callTool("route_task", args, context);
    expect(JSON.parse(result.text)).toEqual({ status: "suggested" });
    expect(apiResponse).toHaveBeenCalledWith("/api/internal/j1-routing", { method: "POST", body: JSON.stringify({ operation: "route_task", arguments: args }) });
    apiResponse.mockClear();
    expect((await callTool("route_task", { ...args, source: "forged" }, context)).isError).toBe(true);
    expect((await callTool("acknowledge_route", { requestId: "stable", kind: "approved" }, context)).isError).toBe(true);
    expect(apiResponse).not.toHaveBeenCalled();
  });
  it("retains tool allow/deny restrictions", () => {
    expect(routingToolAllowed({ allow: [] }, "route_task")).toBe(false);
    expect(routingToolAllowed({ deny: ["mcp:agents:route_task"] }, "route_task")).toBe(false);
    expect(routingToolAllowed({ allow: ["mcp:agents:routing_status"] }, "routing_status")).toBe(true);
    expect(routingToolAllowed(undefined, "unsupported")).toBe(false);
  });
});
